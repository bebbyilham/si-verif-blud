6.3 Model Utama: DokumenPengeluaran

File: app/Models/DokumenPengeluaran.php

class DokumenPengeluaran extends Model
{
    protected $guarded = [];

    // Status Constants
    const STATUS_DIAJUKAN = 'diajukan';
    const STATUS_DIVERIFIKASI = 'diverifikasi';
    const STATUS_DIKEMBALIKAN = 'dikembalikan';
    const STATUS_DISAHKAN = 'disahkan';
    const STATUS_DIBAYAR = 'dibayar';
    const STATUS_DIARSIPKAN = 'diarsipkan';

    protected $casts = [
        'file_path' => 'array',
        'nominal' => 'float',
    ];

    // Auto-generate kode dokumen: BLUD-2026-08-0001
    protected static function booted(): void
    {
        static::creating(function ($model) {
            if (empty($model->kode_dokumen)) {
                $lastDoc = static::latest('id')->first();
                $nextId = $lastDoc ? $lastDoc->id + 1 : 1;
                $model->kode_dokumen = 'BLUD-' . date('Y-m') . '-'
                    . str_pad((string) $nextId, 4, '0', STR_PAD_LEFT);
            }
        });

        // Notifikasi ke Verifikator saat dokumen baru diajukan
        static::created(function ($model) {
            if (!auth()->check()) return;
            $verifikators = User::role('verifikator')->get();
            if ($verifikators->isNotEmpty()) {
                \Filament\Notifications\Notification::make()
                    ->title('Dokumen Baru Diajukan')
                    ->body('Dokumen ' . $model->kode_dokumen . ' diajukan.')
                    ->success()
                    ->sendToDatabase($verifikators);
            }
        });
    }

    // RELASI
    public function bidang()          { return $this->belongsTo(Bidang::class); }
    public function pptk()            { return $this->belongsTo(User::class, 'pptk_id'); }
    public function verifikasis()     { return $this->hasMany(Verifikasi::class, 'dokumen_id'); }
    public function riwayatKoreksis() { return $this->hasMany(RiwayatKoreksi::class, 'dokumen_id'); }
    public function pengesahans()     { return $this->hasMany(Pengesahan::class, 'dokumen_id'); }
    public function pembayarans()     { return $this->hasMany(Pembayaran::class, 'dokumen_id'); }
}

class User extends Authenticatable implements FilamentUser
{
    use HasFactory, Notifiable, HasRoles;

    // Kontrol akses panel — hanya user aktif
    public function canAccessPanel(Panel $panel): bool
    {
        return (bool) $this->status_aktif;

13.2 Audit Trail Observer

File: app/Observers/AuditTrailObserver.php

class AuditTrailObserver
{
    public function created(Model $model): void
    {
        if ($model instanceof DokumenPengeluaran) {
            $this->recordAudit('Dokumen Diajukan - Status: ' . $model->status, ...);
        } elseif ($model instanceof Verifikasi) {
            $this->recordAudit('Verifikasi: ' . $model->hasil . ' ...', ...);
        } // ... dan seterusnya
    }

    public function updated(Model $model): void { /* Record perubahan status */ }
    public function deleted(Model $model): void { /* Record penghapusan */ }

    private function recordAudit(string $aksi, string $tabel, int $id): void
    {
        AuditTrail::create([
            'user_id'           => auth()->id(),
            'aksi'              => mb_substr($aksi, 0, 100),
            'tabel_terdampak'   => $tabel,
            'id_data_terdampak' => $id,
            'waktu'             => now(),
            'ip_address'        => request()?->ip(),
        ]);
    }
}

16.6 Membuat File .cpanel.yml (Auto-Deploy Script)

File ini mengontrol apa yang dijalankan cPanel setiap kali git push / deploy dilakukan. Buat di root proyek lokal:

# File: .cpanel.yml
# Letakkan di root proyek (sejajar dengan composer.json)

deployment:
  tasks:
    # 1. Copy folder public ke public_html
    - export DEPLOYPATH=/home/besignla/public_html
    - /bin/cp -R /home/besignla/repositories/si-verif-blud/public/* $DEPLOYPATH/

    # 2. Buat index.php di public_html yang mengarah ke repo
    - |
      cat > $DEPLOYPATH/index.php << 'ENDOFFILE'
      <?php
      // Redirect semua request ke folder proyek Laravel
      require __DIR__.'/../repositories/si-verif-blud/public/index.php';
      ENDOFFILE

    # 3. Install Composer dependencies (tanpa dev)
    - cd /home/besignla/repositories/si-verif-blud && /usr/local/bin/php /usr/local/bin/composer install --no-dev --optimize-autoloader --no-interaction

    # 4. Cache config, routes, views, icons
    - cd /home/besignla/repositories/si-verif-blud && /usr/local/bin/php artisan config:cache
    - cd /home/besignla/repositories/si-verif-blud && /usr/local/bin/php artisan route:cache
    - cd /home/besignla/repositories/si-verif-blud && /usr/local/bin/php artisan view:cache
    - cd /home/besignla/repositories/si-verif-blud && /usr/local/bin/php artisan icons:cache

    # 5. Jalankan database migration
    - cd /home/besignla/repositories/si-verif-blud && /usr/local/bin/php artisan migrate --force

    # 6. Buat storage symlink
    - cd /home/besignla/repositories/si-verif-blud && /usr/local/bin/php artisan storage:link
